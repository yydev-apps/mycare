Setup Apps Script (Khas Google Sheets)

Ikuti langkah mudah ini untuk menjadikan Google Sheets anda sebagai pangkalan data MyCare:

  1. Buka fail Google Sheets baharu di Google Drive anda.
  2. Pergi ke menu Extensions > Apps Script.
  3. Padamkan semua kod sedia ada dan gantikan dengan kod di bawah:
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var recordSheet = ss.getSheetByName("Records") || ss.insertSheet("Records");
  var profileSheet = ss.getSheetByName("Profile") || ss.insertSheet("Profile");
  
  var profileData = {};
  var pValues = profileSheet.getDataRange().getValues();
  if (pValues.length > 1) {
    var lastRow = pValues[pValues.length - 1];
    profileData = {
      name: lastRow[0] || '',
      age: lastRow[1] || '',
      gender: lastRow[2] || 'Lelaki',
      weight: lastRow[3] || '',
      height: lastRow[4] || ''
    };
  }
  
  var records = [];
  var rValues = recordSheet.getDataRange().getValues();
  for (var i = 1; i < rValues.length; i++) {
    var row = rValues[i];
    if (row[0]) {
      records.push({
        id: String(row[0]),
        date: row[1],
        type: row[2],
        valueStr: row[3],
        extraStr: row[4],
        status: row[5],
        notes: row[6],
        colorClass: row[7]
      });
    }
  }
  
  return ContentService
    .createTextOutput(JSON.stringify({ profile: profileData, records: records }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  
  if (data.action === 'SAVE_PROFILE') {
    var profileSheet = ss.getSheetByName("Profile") || ss.insertSheet("Profile");
    if (profileSheet.getLastRow() === 0) {
      profileSheet.appendRow(["Nama", "Umur", "Jantina", "Berat", "Tinggi", "Tarikh"]);
    }
    profileSheet.appendRow([
      data.name || '',
      data.age || '',
      data.gender || '',
      data.weight || '',
      data.height || '',
      new Date().toLocaleString('ms-MY')
    ]);
  } else if (data.action === 'ADD_RECORD') {
    var recordSheet = ss.getSheetByName("Records") || ss.insertSheet("Records");
    if (recordSheet.getLastRow() === 0) {
      recordSheet.appendRow(["ID", "Tarikh", "Jenis", "Bacaan", "Butiran", "Status", "Nota", "ColorClass"]);
    }
    recordSheet.appendRow([
      data.id || Date.now().toString(),
      data.date || new Date().toISOString(),
      data.type || '',
      data.valueStr || '',
      data.extraStr || '',
      data.status || '',
      data.notes || '',
      data.colorClass || ''
    ]);
  } else if (data.action === 'DELETE_RECORD') {
    var recordSheet = ss.getSheetByName("Records");
    if (recordSheet) {
      var rows = recordSheet.getDataRange().getValues();
      for (var i = rows.length - 1; i >= 1; i--) {
        if (String(rows[i][0]) === String(data.id)) {
          recordSheet.deleteRow(i + 1);
          break;
        }
      }
    }
  }
  
  return ContentService
    .createTextOutput(JSON.stringify({ status: "SUCCESS" }))
    .setMimeType(ContentService.MimeType.JSON);
}
